== Quick Start ==

  # Register Engine A with Restate
  kubectl exec -it deploy/{{ .Release.Name }}-restate -- \
    curl -X POST http://localhost:{{ .Values.restate.adminPort }}/deployments \
      -H 'content-type: application/json' \
      -d '{"uri": "http://{{ .Release.Name }}-engine-a:{{ .Values.engineA.port }}/restate"}'

  # Verify BPMN catalog table
  kubectl exec -it {{ .Release.Name }}-postgresql-0 -- \
    psql -U {{ .Values.postgresql.auth.username }} -d {{ .Values.postgresql.auth.database }} \
    -c "SELECT * FROM bpmn_catalog;"
