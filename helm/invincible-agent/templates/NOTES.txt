== Quick Start ==

  # Register Engine A with Restate (must run curl against Restate admin port externally if Restate is not local)
  kubectl exec -it deploy/{{ .Release.Name }}-engine-a -- \
    curl -X POST {{ include "invincible-agent.restateAdminUrl" . }}/deployments \
      -H 'content-type: application/json' \
      -d '{"uri": "http://{{ .Release.Name }}-engine-a:{{ .Values.engineA.port }}/restate"}'

  # Verify BPMN catalog table is accessible (can use external client if BYO Postgres)
  kubectl exec -it {{ .Release.Name }}-engine-a -- \
    psql -h {{ include "invincible-agent.pgHost" . }} -p {{ include "invincible-agent.pgPort" . }} -U {{ include "invincible-agent.pgUser" . }} -d {{ include "invincible-agent.pgDatabase" . }} \
    -c "SELECT * FROM bpmn_catalog;"
